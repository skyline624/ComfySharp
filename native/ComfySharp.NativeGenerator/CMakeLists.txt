cmake_minimum_required(VERSION 3.24)
project(ComfySharpNativeGenerator LANGUAGES CXX)
find_package(Torch 2.10.0 EXACT REQUIRED)
add_library(ComfySharp.Native SHARED generator.cpp)
target_compile_features(ComfySharp.Native PRIVATE cxx_std_17)
target_include_directories(ComfySharp.Native PRIVATE ${TORCH_INCLUDE_DIRS})
target_link_libraries(ComfySharp.Native PRIVATE ${TORCH_LIBRARIES})
set_property(TARGET ComfySharp.Native PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
